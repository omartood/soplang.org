<SoplangCodeWindow title="keywords_quick_examples.sop" code={`// Variables
door magac = "Sharafdin"
madoor PI = 3.14159
abn da = 25
jajab qiimo = 3.14
qoraal qora = "Salaan"
bool sax = run
teed numbers = [1, 2, 3]
walax qof = { magac: "Ayaan" }
door wax = null

// Control flow
haddii (da >= 18) { qor("Qof weyn") } ugudambeyn { qor("Ilmo") }

dooro (qora) {
  xaalad "Salaan" { qor("Hello!") }
  ugudambeyn { qor("Unknown") }
}

kuceli (i 1 ilaa 5 :: 2) { qor(i) }

abn x = 1
intay (x < 3) {
  qor(x)
  x = x + 1
}

// Functions
hawl isuGee(a, b) { celi a + b }
qor(isuGee(2, 3))

// Classes
qaab Qof {
  hawl dhaw(magac) { nafta.magac = magac }
  hawl salaam() { celi "Salaan, " + nafta.magac }
}
door q = cusub Qof("Aamina")
qor(q.salaam())

// Error handling
fasax {
  qor(10 / 0)
} qabo (khalad) {
  qor("Khalad: " + khalad)
}`}/>
